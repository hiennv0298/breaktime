```typescript
// src/logic/waypointWalker.ts
export interface WaypointPoint { x: number; z: number; dwellSec: number }
export interface WalkerState { index: number; x: number; z: number; mode: 'walk' | 'dwell'; dwellLeft: number; frozen: boolean }
export function createWalker(points: WaypointPoint[], start: { x: number; z: number }, startIndex?: number): WalkerState;
export function stepWalker(s: WalkerState, points: WaypointPoint[], dt: number, speed?: number, arriveRadius?: number):
  { state: WalkerState; vx: number; vz: number; facingYaw: number };   // speed default 1.4 m/s, arriveRadius 0.15 m; facingYaw = atan2(dx, dz)
export function nearestIndex(points: WaypointPoint[], x: number, z: number): number;
// src/render/characters.ts
export const CHARACTER_PARTS: readonly ['leg-left', 'leg-right', 'torso', 'arm-left', 'arm-right', 'head'];
export interface CharacterAsset { scene: Object3D; clips: Record<string, AnimationClip> }
export function parseCharacterAsset(glb: ArrayBuffer): Promise<CharacterAsset>;
export interface CharacterInstance { root: Object3D; mixer: AnimationMixer; parts: Object3D[]; material: MeshLambertMaterial; setMotion(name: 'idle' | 'walk' | 'attack-melee-right' | 'interact-right', fadeSec?: number): void; dispose(): void }
export function spawnCharacter(asset: CharacterAsset, textureLetter: string): CharacterInstance;   // letters a..r
// src/game/waypoints.ts
export const NPC_ROUTES: WaypointPoint[][];   // 3 routes
// src/game/npc.ts
export interface Npc { id: string; character: CharacterInstance; body: RigidBody; walker: WalkerState; fixedUpdate(dt: number): void; frameUpdate(dt: number): void; pos(): { x: number; y: number; z: number } }
export function createNpc(ctx: GameCtx, opts: { id: string; asset: CharacterAsset; texture: string; route: WaypointPoint[]; startIndex: number }): Npc;
```
