```typescript
// src/logic/temper.ts (pure)
export type Temper = 'hot' | 'normal' | 'calm';
export const TEMPERS: readonly Temper[];              // ['hot', 'normal', 'calm']
export const DEFAULT_TEMPER: Temper;                  // 'normal'
export const TEMPER_LABEL_VI: Readonly<Record<Temper, string>>; // hot 'Nóng', normal 'Thường', calm 'Hiền'
export const SLAPS_TO_ANGER: Readonly<Record<Temper, number>>;  // hot 1, normal 2, calm 3
export function isTemper(v: unknown): v is Temper;

// src/logic/anger.ts (pure)
export const ANGER_MAX = 100;
export const ANGER_THRESHOLD = 100;
export const SLAP_ANGER: Readonly<Record<Temper, number>>; // hot 100, normal 60, calm 42 (never early with jitter 0..5: normal max 65 after 1 slap, calm max 94 after 2)
export const SLAP_JITTER_MAX = 5;      // integer jitter 0..5 added per slap
export const DECAY_DELAY_SEC = 6;
export const DECAY_PER_SEC = 8;
export const CALM_BELOW = 40;
export const REACT_MIN_SEC = 0.2;
export const REACT_MAX_SEC = 0.5;
export interface AngerState { value: number; sinceSlapSec: number; slaps: number }
export function createAnger(): AngerState;
export function onSlapped(s: AngerState, temper: Temper, rng: () => number, opts?: { jitter?: boolean }): AngerState;
export function tickAnger(s: AngerState, dt: number, opts?: { holdDecay?: boolean }): AngerState; // holdDecay true (NPC ragdoll / recover): neither sinceSlapSec nor value changes
export function isAngry(s: AngerState): boolean;   // value >= ANGER_THRESHOLD
export function isCalm(s: AngerState): boolean;    // value < CALM_BELOW
export function satisfyAnger(s: AngerState): AngerState; // value 0, sinceSlapSec kept, slaps kept
export function reactionDelaySec(rng: () => number): number; // REACT_MIN_SEC + rng() * (REACT_MAX_SEC - REACT_MIN_SEC)
export function fightFromQuery(search: string): 'always' | null;
export function effectiveTemper(t: Temper, fight: 'always' | null): Temper; // 'hot' when fight === 'always'

// src/logic/attackTokens.ts (pure)
export const MAX_PURSUERS = 3;
export const MAX_ATTACKERS = 1;
export interface Contender { id: string; anger: number; dist: number; wantsPursue: boolean; wantsStrike: boolean }
export interface TokenHoldings { pursue: ReadonlySet<string>; strike: ReadonlySet<string> }
export function emptyHoldings(): TokenHoldings;
export function arbitrate(contenders: readonly Contender[], held: TokenHoldings): { pursue: Set<string>; strike: Set<string> };
```
