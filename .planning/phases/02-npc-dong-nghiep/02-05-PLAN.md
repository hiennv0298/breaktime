The game side (plan 02-10) only feeds observations and applies commands.
Output: src/logic/combatFsm.ts, src/logic/combatDirector.ts, seedFor in src/logic/rng.ts, unit tests.
</objective>

export type CombatState = 'routine' | 'down' | 'fume' | 'pursue' | 'windup' | 'cooldown' | 'return';
export type CombatMove = 'walker' | 'hold' | 'pursue' | 'sidestep' | 'return';
export type CombatFsmEvent = 'none' | 'down' | 'interrupted' | 'fume' | 'pursue' | 'windup' | 'strike' | 'give-up' | 'resume';
export interface CombatFsmState { state: CombatState; reactLeft: number; chaseSec: number; farSec: number; windowSec: number; windowStartDist: number; stuck: number; sidestepLeft: number; sidestepSign: 1 | -1; windupLeft: number; cooldownLeft: number; grudgeSatisfied: boolean }

export interface CombatNpcObs { id: string; memberId: string; temper: Temper; x: number; z: number; yaw: number; physics: 'animated' | 'ragdoll' | 'recover'; routeX: number; routeZ: number }

export interface CombatCommand { id: string; state: CombatState; move: CombatMove; stepX: number; stepZ: number; faceX: number; faceZ: number; motion: 'idle' | 'walk' | 'sprint' | 'emote-no' | 'attack-melee-right' | null; angry: boolean; marker: boolean; token: 'pursue' | 'strike' | null; anger: number }

export interface CombatStepResult { commands: CombatCommand[]; events: CombatEvent[] }
export interface CombatSnapshot { strikes: number; landed: number; missed: number; interrupted: number; giveUps: number; maxPursuers: number; maxAttackers: number; pursuers: string[]; attackers: string[]; npcs: Array<{ id: string; memberId: string; temper: Temper; state: CombatState; anger: number; token: 'pursue' | 'strike' | null }> }
